document.addEventListener('click', function (e) {
  const btn = e.target.closest('.gs-expand-all');
  if (!btn) return;

  const expanding = btn.getAttribute('aria-expanded') !== 'true';

  document.querySelectorAll('.gs-accordion-item__title').forEach(title => {
    const item = title.closest('.gs-accordion-item');
    const isOpen = title.getAttribute('aria-expanded') === 'true' ||
                   (item && item.classList.contains('gsopen'));
    if (isOpen !== expanding) title.click();
  });

  btn.setAttribute('aria-expanded', expanding);
  btn.textContent = expanding ? 'Collapse all' : 'Expand all';
});
</script>
